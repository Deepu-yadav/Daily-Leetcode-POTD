class Solution {
    public List<Integer> findAllPeople(int n, int[][] meetings, int firstPerson) {
    
    Map<Integer,List<int[]>>map=new TreeMap<>();
    boolean[]knowSecret=new boolean[n];
    knowSecret[0]=true;
    knowSecret[firstPerson]=true;

     for(int[]arr:meetings){
        int u=arr[0];
        int v=arr[1];
        int time=arr[2];
        map.computeIfAbsent(time,k->new ArrayList<>()).add(new int[]{u,v});
     }



     for(int key:map.keySet()){
     List<int[]>list=map.get(key);

     HashMap<Integer,List<Integer>>adj=new HashMap<>();
     Queue<Integer>q=new LinkedList<>();
     Set<Integer>set=new HashSet<>();

     for(int[]a:list){
       int p1=a[0];
       int p2=a[1];

       if(knowSecret[p1]==true && !set.contains(p1)){
          q.add(p1);
          set.add(p1);
       }
        if(knowSecret[p2]==true && !set.contains(p2)){
          q.add(p2);
         set.add(p2);
       }
       adj.computeIfAbsent(p1,k->new ArrayList<>()).add(p2);
       adj.computeIfAbsent(p2,k->new ArrayList<>()).add(p1);
     }



          while(!q.isEmpty()){
           
           int person1=q.poll();

           for(int v:adj.getOrDefault(person1,Collections.emptyList())){
               if(knowSecret[v]==false){
                knowSecret[v]=true;
                q.add(v);
               }
           }
          }
     }
     List<Integer>ans=new ArrayList<>();
     
     List<Integer>list=new ArrayList<>();
     for(int i=0; i<n; i++){
        if(knowSecret[i]==true){
            ans.add(i);
        }
     }
     return ans;
    }
}
