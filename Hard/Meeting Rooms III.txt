class Solution {
    public int mostBooked(int n, int[][] meetings) {

        // Sort meetings by start time
        Arrays.sort(meetings, (a, b) -> Integer.compare(a[0], b[0]));

        // Free rooms (min room index)
        PriorityQueue<Integer> freeRooms = new PriorityQueue<>();

        // Busy rooms: [roomNumber, endTime]
        PriorityQueue<long[]> busyRooms = new PriorityQueue<>(
            (a, b) -> a[1] != b[1]
                ? Long.compare(a[1], b[1])
                : Long.compare(a[0], b[0])
        );

        // Count meetings per room
        int[] count = new int[n];

        // Initially all rooms are free
        for (int i = 0; i < n; i++) {
            freeRooms.offer(i);
        }

        // Process each meeting
        for (int[] meeting : meetings) {
            long start = meeting[0];
            long end = meeting[1];

            // Free all rooms that have completed before current meeting
            while (!busyRooms.isEmpty() && busyRooms.peek()[1] <= start) {
                freeRooms.offer((int) busyRooms.poll()[0]);
            }

            if (!freeRooms.isEmpty()) {
                // Assign smallest free room
                int room = freeRooms.poll();
                count[room]++;
                busyRooms.offer(new long[]{room, end});
            } else {
                // Delay meeting: use earliest ending room
                long[] earliest = busyRooms.poll();
                int room = (int) earliest[0];
                long newEnd = earliest[1] + (end - start);

                count[room]++;
                busyRooms.offer(new long[]{room, newEnd});
            }
        }

        // Find room with max meetings
        int ans = 0;
        for (int i = 1; i < n; i++) {
            if (count[i] > count[ans]) {
                ans = i;
            }
        }

        return ans;
    }
}
