class Solution {
    String result="";
    public String lexGreaterPermutation(String s, String target) {

        int[]count=new int[32];
        for(int i=0; i<s.length(); i++){
            char ch=s.charAt(i);
            count[ch-'a']++;
        }
        solve(new StringBuilder(),0,count,target,false,s);
        return result;

    }

    public boolean solve(StringBuilder curr,int index,int[] count,String target, boolean isTrue,String s){
             
             if(index==target.length()){
              if(isTrue){
                result=curr.toString();
                  return true;
              }
              return false;
            
             }

             for(char ch='a'; ch<='z'; ch++){
                char c=target.charAt(index);
               if(isTrue==false && ch<c)continue;
               if(count[ch-'a']==0)continue;
               curr.append(ch);
               count[ch-'a']--;
               boolean flag=(isTrue==true || ch>c);
               if(solve(curr,index+1,count,target,flag,s)){
                return true;
               }
               count[ch-'a']++;
               curr.deleteCharAt(curr.length()-1);
             }
             return false;

    }
}
