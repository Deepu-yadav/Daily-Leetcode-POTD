class Solution {
    public int minDeletionSize(String[] strs) {
          int rows=strs.length; 
          boolean[]alreadySorted=new boolean[rows-1];
          int count=0;


        for(int col=0; col<strs[0].length(); col++){
            boolean isDeletion=false;

            for(int row=0; row<rows-1; row++){
                if(!alreadySorted[row] && strs[row].charAt(col)>strs[row+1].charAt(col)){
                    isDeletion=true;
                    break;
                }
            }
            if(isDeletion){
                count++;
                continue;
            }

            // mark all the pair true
         for(int row=0; row<rows-1; row++){
             if(!alreadySorted[row] && strs[row+1].charAt(col)>strs[row].charAt(col)){
              alreadySorted[row]=true;  
             }
        }
        }

        return count;  
    }
}
