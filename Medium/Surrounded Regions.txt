class Solution {
    public void solve(char[][] board) {
        
        int row=board.length;
        int col=board[0].length;
        boolean[][]temp=new boolean[row][col];

        boolean[][]visited=new boolean[row][col];
          // first row
        for(int i=0; i<col; i++){
            if(!visited[0][i] && board[0][i]=='O'){
                dfs(board,visited,0,i,temp);
            }
        }

        // last row.
        for(int i=1; i<col; i++){
            if(!visited[row-1][i] && board[row-1][i]=='O'){
                dfs(board,visited,row-1,i,temp);
            }
        }


        // first col

          for(int i=1; i<row; i++){
            if(!visited[i][0] && board[i][0]=='O'){
                dfs(board,visited,i,0,temp);
            }
        }
        // left last col

        for(int i=1; i<row; i++){
            if(!visited[i][col-1] && board[i][col-1]=='O'){
                dfs(board,visited,i,col-1,temp);
            }
        }

        for(int i=0; i<row; i++){
            for(int j=0; j<col; j++){
                if(!temp[i][j] && board[i][j]=='O'){
                    board[i][j]='X';
                }
            }
        }



    }

    public void dfs(char[][]board,boolean[][]visited,int i,int j,boolean[][]temp){
        if(i<0 || j<0 || i>=board.length || j>=board[0].length || visited[i][j] || temp[i][j] || board[i][j]=='X'){
            return;
        }
        temp[i][j]=true;
        visited[i][j]=true;

        dfs(board,visited,i-1,j,temp);
        dfs(board,visited,i+1,j,temp);
        dfs(board,visited,i,j-1,temp);
        dfs(board,visited,i,j+1,temp);
        visited[i][j]=true;
    }
}
