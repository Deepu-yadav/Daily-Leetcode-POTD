class Solution {
    public int maximizeSquareHoleArea(int n, int m, int[] hBars, int[] vBars) {

     int maximumConsecutiveV=1;
     int maximumConsecutiveH=1;
     int longestConsecutiveV=1;
     int longestConsecutiveH=1;
       Arrays.sort(hBars);
       Arrays.sort(vBars);
    


    for(int i=1; i<hBars.length; i++){
           if(hBars[i]-hBars[i-1]==1){
            longestConsecutiveH++;
           }
           else{
            longestConsecutiveH=1;
           }
            maximumConsecutiveH=Math.max(maximumConsecutiveH, longestConsecutiveH);
    }




    for(int i=1; i<vBars.length; i++){
      
      if(vBars[i]-vBars[i-1]==1){
         longestConsecutiveV++;
      }
      else{
         longestConsecutiveV=1;
      }
       maximumConsecutiveV=Math.max(maximumConsecutiveV,longestConsecutiveV);
    }

    int side=Math.min(maximumConsecutiveV,maximumConsecutiveH)+1;
    return side*side;

    }
}
