class Solution {

    static final long NEG_INF = (long) -1e18;
    long[][][] dp;

    public long maximumProfit(int[] prices, int k) {
        int n = prices.length;
        dp = new long[n][k + 1][3];

        for (int i = 0; i < n; i++) {
            for (int j = 0; j <= k; j++) {
                for (int c = 0; c < 3; c++) {
                    dp[i][j][c] = NEG_INF;
                }
            }
        }

        return solve(prices, 0, 0, k);
    }

    public long solve(int[] prices, int index, int state, int k) {

        if (index == prices.length) {
            return (state == 0) ? 0 : NEG_INF;
        }

        if (dp[index][k][state] != NEG_INF) {
            return dp[index][k][state];
        }

        long skip = solve(prices, index + 1, state, k);
        long take = NEG_INF;

        if (state == 0) {
            // open long or short
            take = Math.max(
                -prices[index] + solve(prices, index + 1, 1, k),
                 prices[index] + solve(prices, index + 1, 2, k)
            );
        }
        else if (state == 1 && k > 0) {
            // close long
            take = prices[index] + solve(prices, index + 1, 0, k - 1);
        }
        else if (state == 2 && k > 0) {
            // close short
            take = -prices[index] + solve(prices, index + 1, 0, k - 1);
        }

        return dp[index][k][state] = Math.max(skip, take);
    }
}
