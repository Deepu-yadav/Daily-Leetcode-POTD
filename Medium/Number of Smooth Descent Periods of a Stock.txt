class Solution {
    public long getDescentPeriods(int[] prices) {

     long ans=1;
     int i=0;
     int j=1;
     while(j<prices.length){
         if((prices[j-1]-1)==(prices[j])){
             ans+=(j-i+1);
            j++;
          }
         else{
            i=j;
            ans+=(j-i+1);
              j++;
         }
        
      }
      return ans;  
    }
}
