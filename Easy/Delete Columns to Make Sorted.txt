class Solution {
    public int minDeletionSize(String[] strs) {
        int count=0;
        for(int i=0; i<strs[0].length(); i++){
             boolean isDeletion=false;
             char prev=' ';
            for(int j=0; j<strs.length; j++){
               char curr=strs[j].charAt(i);
               if(curr>=prev){
                prev=curr;
               }
               else{
                isDeletion=true;
                break;
               }
            }
            if(isDeletion)count++;
        }
        return count;
    }
}
