class Solution {
    public int missingMultiple(int[] nums, int k) {
        
        HashMap<Integer,Integer>map=new HashMap<>();

        for(int i=0; i<nums.length; i++){
            map.put(nums[i],1);
        }
        int multiple=k;
        while(true){
            if(!map.containsKey(multiple))return multiple;
            multiple+=k;
        }
    }
}
